server : Mcp.Server(McpHttp.Caller(Context))

	name: "notes",
	version: "1.0.0",
	tools: [

		McpHttp.endpoint(Api.get_api_notes_id, { name: "read_note", description: "A note, by its id" }, McpHttp.openapi({})),
		McpHttp.function(Echo.echo!, { name: "echo", description: "Says a message back" }, McpHttp.openapi({})),
		McpHttp.tool({ name: "greet", description: "Greets someone" }, greet!, McpHttp.openapi({})),
	],
}
